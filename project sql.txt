CREATE TABLE category (
  cat_id NUMBER PRIMARY KEY,
  cat_name VARCHAR2(50)
);

CREATE TABLE products (
  prod_id NUMBER PRIMARY KEY,
  prod_name VARCHAR2(50),
  brand VARCHAR2(50),
  price NUMBER,
  cat_id NUMBER REFERENCES category(cat_id)
);
